using System;
using System.Windows;
using System.Windows.Threading;
using Microsoft.Extensions.Configuration;
using Microsoft.Extensions.DependencyInjection;
using MallClient.Interfaces;
using MallClient.Models;
using MallClient.Providers;
using MallClient.Schemas;
using MallClient.Services;
using MallClient.ViewModels;
using MallClient.Views;

namespace MallClient;

public partial class App : Application {
    private const string AppSettingsFile = "appsettings.json";
    private const string ApiSettingsSection = "ApiSettings";
    private const string ConfigErrorTitle = "Ошибка конфигурации";
    private const string ConfigErrorMessage = "Не найден BaseUrl в appsettings.json";

    public static IServiceProvider Services { get; private set; } = null!;

    protected override void OnStartup(StartupEventArgs e) {
        base.OnStartup(e);

        DispatcherUnhandledException += OnDispatcherUnhandledException;
        AppDomain.CurrentDomain.UnhandledException += OnDomainUnhandledException;
        AppDomain.CurrentDomain.ProcessExit += OnProcessExit;
        TaskScheduler.UnobservedTaskException += OnUnobservedTaskException;
        SessionEnding += OnSessionEnding;

        var configuration = LoadConfiguration();
        var apiSettings = configuration.GetSection(ApiSettingsSection).Get<AppSettings>();

        if (apiSettings == null || string.IsNullOrEmpty(apiSettings.BaseUrl)) {
            ShowFatalError(ConfigErrorMessage, ConfigErrorTitle);
            return;
        }

        Services = ConfigureServices(apiSettings);

        var session = Services.GetRequiredService<ISession>();
        session.SessionExpired += OnSessionExpired;

        var windowService = Services.GetRequiredService<IWindowService>();
        windowService.ShowLoginWindow();
    }

    private void OnSessionExpired() {
        Current?.Dispatcher?.Invoke(() => {
            try {
                var auth = Services.GetRequiredService<IAuthService>();
                var windowService = Services.GetRequiredService<IWindowService>();

                if (Current.MainWindow is LoginWindow) return;

                auth.Logout();
                windowService.ShowLoginWindow();

                MessageBox.Show("Ваш аккаунт был удален или срок действия сессии истек.\nПожалуйста, авторизуйтесь заново.",
                    "Доступ запрещен", MessageBoxButton.OK, MessageBoxImage.Warning);
            } catch {
            }
        });
    }

    private void OnDispatcherUnhandledException(object sender, DispatcherUnhandledExceptionEventArgs e) {
        MessageBox.Show(e.Exception.ToString(), "Необработанная ошибка", MessageBoxButton.OK, MessageBoxImage.Error);
        e.Handled = true;
    }

    private void OnDomainUnhandledException(object sender, UnhandledExceptionEventArgs e) {
        var ex = e.ExceptionObject as Exception;
        MessageBox.Show(ex?.ToString() ?? "Неизвестная ошибка", "Критическая ошибка", MessageBoxButton.OK, MessageBoxImage.Error);
    }

    private void OnUnobservedTaskException(object? sender, UnobservedTaskExceptionEventArgs e) {
        e.SetObserved();
    }

    private void OnSessionEnding(object sender, SessionEndingCancelEventArgs e) {
        ForceExit();
    }

    private void OnProcessExit(object? sender, EventArgs e) {
        DisposeServices();
    }

    protected override void OnExit(ExitEventArgs e) {
        DisposeServices();
        base.OnExit(e);
    }

    public static void ForceExit() {
        try { DisposeServices(); } catch { }
        try { Current?.Shutdown(0); } catch { }
        Environment.Exit(0);
    }

    private static void DisposeServices() {
        if (Services is IDisposable disposable) {
            try { disposable.Dispose(); } catch { }
        }
    }

    private IConfiguration LoadConfiguration() {
        return new ConfigurationBuilder()
            .SetBasePath(AppContext.BaseDirectory)
            .AddJsonFile(AppSettingsFile, optional: false, reloadOnChange: true)
            .Build();
    }

    private IServiceProvider ConfigureServices(AppSettings apiSettings) {
        var services = new ServiceCollection();

        services.AddHttpClient("MallClient", c => {
            c.BaseAddress = new Uri(apiSettings.BaseUrl);
            c.Timeout = TimeSpan.FromSeconds(30);
        })
        .AddHttpMessageHandler<AuthMessageHandler>();

        services.AddTransient<AuthMessageHandler>();

        services.AddSingleton<ISession, Session>();
        services.AddSingleton<ITokenStore, TokenStore>();
        services.AddSingleton<IApiClientFactory, ApiClientFactory>();
        services.AddSingleton<IFieldTranslator, FieldTranslator>();
        services.AddSingleton<IApiErrorParser, ApiErrorParser>();
        services.AddSingleton<IAuthApi, AuthApi>();
        services.AddSingleton<IAuthService, AuthService>();

        services.AddSingleton<IPremiseService, PremiseService>();
        services.AddSingleton<ITenantService, TenantService>();
        services.AddSingleton<IContractService, ContractService>();
        services.AddSingleton<IPaymentService, PaymentService>();
        services.AddSingleton<IRequestService, RequestService>();
        services.AddSingleton<IUserService, UserService>();
        services.AddSingleton<IWindowService, WindowService>();

        services.AddSingleton<IInputValidator, InputValidator>();
        services.AddSingleton<PayloadBuilder>();
        services.AddSingleton<SchemaFactory>();

        services.AddTransient<PremiseOptionsProvider>();
        services.AddTransient<TenantOptionsProvider>();
        services.AddTransient<ContractOptionsProvider>();

        services.AddTransient<LoginViewModel>();
        services.AddTransient<RegisterViewModel>();
        services.AddTransient<MainViewModel>();

        services.AddTransient<LoginWindow>();
        services.AddTransient<RegisterWindow>();
        services.AddTransient<MainWindow>();
        services.AddTransient<GenericEditWindow>();

        return services.BuildServiceProvider();
    }

    private void ShowFatalError(string message, string title) {
        MessageBox.Show(message, title, MessageBoxButton.OK, MessageBoxImage.Error);
        ForceExit();
    }
}

using System;
using System.Globalization;
using System.Windows.Data;

namespace MallClient.Converters;

public class RoleToRussianConverter : IValueConverter {
    public object Convert(object value, Type targetType, object parameter, CultureInfo culture) {
        var role = value as string;
        return role switch {
            "admin" => "Администратор",
            "manager" => "Менеджер",
            "accountant" => "Бухгалтер",
            "technician" => "Техник",
            "support" => "Поддержка",
            _ => role ?? string.Empty
        };
    }

    public object ConvertBack(object value, Type targetType, object parameter, CultureInfo culture) {
        throw new NotSupportedException();
    }
}

using System;
using System.Globalization;
using System.Windows;
using System.Windows.Data;

namespace MallClient.Converters;

public class StringToVisibilityConverter : IValueConverter {
    public object Convert(object value, Type targetType, object parameter, CultureInfo culture) {
        var s = value as string;
        return string.IsNullOrWhiteSpace(s) ? Visibility.Collapsed : Visibility.Visible;
    }

    public object ConvertBack(object value, Type targetType, object parameter, CultureInfo culture) {
        throw new NotSupportedException();
    }
}

using System.ComponentModel;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Input;
using MallClient.Models;
using MallClient.ViewModels;

namespace MallClient.Views;

public partial class MainWindow : Window {
    private MainViewModel? _viewModel;
    private bool _initialized;

    public MainWindow(MainViewModel viewModel) {
        _viewModel = viewModel;
        InitializeComponent();
        DataContext = _viewModel;
    }

    private async void MainWindow_OnLoaded(object sender, RoutedEventArgs e) {
        if (_viewModel == null) return;

        await _viewModel.LoadDataAsync();

        _initialized = true;

        if (TabPremises.IsVisible) TabPremises.IsChecked = true;
        else if (TabTenants.IsVisible) TabTenants.IsChecked = true;
        else if (TabContracts.IsVisible) TabContracts.IsChecked = true;
        else if (TabPayments.IsVisible) TabPayments.IsChecked = true;
        else if (TabRequests.IsVisible) TabRequests.IsChecked = true;
        else if (TabUsers.IsVisible) TabUsers.IsChecked = true;
    }

    private void MainWindow_OnClosing(object? sender, CancelEventArgs e) {
        if (Application.Current.MainWindow == this) {
            App.ForceExit();
        }
    }

    private void TitleBar_MouseLeftButtonDown(object sender, MouseButtonEventArgs e) {
        if (e.ButtonState == MouseButtonState.Pressed)
            DragMove();
    }

    private void CloseButton_Click(object sender, RoutedEventArgs e) {
        App.ForceExit();
    }

    private async void Tab_Checked(object sender, RoutedEventArgs e) {
        if (_viewModel == null) return;
        if (sender is not RadioButton rb) return;

        PanelPremises.Visibility = Visibility.Collapsed;
        PanelTenants.Visibility = Visibility.Collapsed;
        PanelContracts.Visibility = Visibility.Collapsed;
        PanelPayments.Visibility = Visibility.Collapsed;
        PanelRequests.Visibility = Visibility.Collapsed;
        PanelUsers.Visibility = Visibility.Collapsed;

        switch (rb.Name) {
            case nameof(TabPremises):
                PanelPremises.Visibility = Visibility.Visible;
                if (_initialized) await _viewModel.ReloadPremisesAsync();
                break;
            case nameof(TabTenants):
                PanelTenants.Visibility = Visibility.Visible;
                if (_initialized) await _viewModel.ReloadTenantsAsync();
                break;
            case nameof(TabContracts):
                PanelContracts.Visibility = Visibility.Visible;
                if (_initialized) await _viewModel.ReloadContractsAsync();
                break;
            case nameof(TabPayments):
                PanelPayments.Visibility = Visibility.Visible;
                if (_initialized) await _viewModel.ReloadPaymentsAsync();
                break;
            case nameof(TabRequests):
                PanelRequests.Visibility = Visibility.Visible;
                if (_initialized) await _viewModel.ReloadRequestsAsync();
                break;
            case nameof(TabUsers):
                PanelUsers.Visibility = Visibility.Visible;
                if (_initialized) await _viewModel.ReloadUsersAsync();
                break;
        }
    }

using System;
using System.Collections.Generic;

namespace MallClient.Models;

public class FieldItem {
    public string Name { get; set; } = string.Empty;
    public string Label { get; set; } = string.Empty;
    public string Value { get; set; } = string.Empty;

    public bool IsDropdown { get; set; } = false;
    public bool IsNumeric { get; set; } = false;
    public bool IsDate { get; set; } = false;
    public bool IsPhone { get; set; } = false;
    public bool IsEmail { get; set; } = false;

    public int MaxLength { get; set; } = 0;

    public DateTime? DateValue {
        get {
            if (string.IsNullOrWhiteSpace(Value)) return null;
            if (DateTime.TryParse(Value, out var dt)) return dt;
            return null;
        }
        set {
            Value = value?.ToString("yyyy-MM-dd") ?? string.Empty;
        }
    }

    public List<OptionItem> Options { get; set; } = new();
    public bool IsValid { get; set; } = true;
}

public class OptionItem {
    public string Value { get; set; } = string.Empty;
    public string DisplayName { get; set; } = string.Empty;

    public override string ToString() {
        return DisplayName;
    }
}

using System.Collections.Generic;
using System.Text.Json;
using MallClient.Interfaces;
using MallClient.Models;

namespace MallClient.Services;

public class ApiErrorParser : IApiErrorParser {
    private const string DefaultMessage = "Не удалось выполнить операцию.";

    private readonly IFieldTranslator _translator;

    public ApiErrorParser(IFieldTranslator translator) {
        _translator = translator;
    }

    public string Parse(string json) {
        if (string.IsNullOrWhiteSpace(json))
            return DefaultMessage;

        try {
            var payload = JsonSerializer.Deserialize<ApiErrorResponse>(json);
            if (payload == null) return DefaultMessage;

            if (payload.Fields != null && payload.Fields.Count > 0) {
                var parts = new List<string>();
                foreach (var kv in payload.Fields)
                    parts.Add($"{_translator.Translate(kv.Key)}: {kv.Value}");
                return string.Join("\n", parts);
            }

            if (!string.IsNullOrEmpty(payload.Error))
                return payload.Error;
        } catch (JsonException) {
        }

        return DefaultMessage;
    }
}

using System.Net;
using System.Net.Http;
using System.Net.Http.Headers;
using System.Threading;
using System.Threading.Tasks;
using MallClient.Interfaces;

namespace MallClient.Services;

public class AuthMessageHandler : DelegatingHandler {
    private static readonly SemaphoreSlim RefreshLock = new(1, 1);

    private readonly ITokenStore _tokenStore;
    private readonly ISession _session;
    private readonly IAuthApi _authApi;

    public AuthMessageHandler(ITokenStore tokenStore, ISession session, IAuthApi authApi) {
        _tokenStore = tokenStore;
        _session = session;
        _authApi = authApi;
    }

    protected override async Task<HttpResponseMessage> SendAsync(HttpRequestMessage request, CancellationToken cancellationToken) {
        var tokenBefore = _tokenStore.AccessToken;
        if (!string.IsNullOrEmpty(tokenBefore)) {
            request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", tokenBefore);
        }

        var response = await base.SendAsync(request, cancellationToken);

        if (response.StatusCode == HttpStatusCode.Unauthorized &&
            !request.RequestUri!.AbsolutePath.Contains("/api/auth/login") &&
            !request.RequestUri!.AbsolutePath.Contains("/api/auth/refresh")) {

            bool refreshed;

            await RefreshLock.WaitAsync(cancellationToken);
            try {
                if (_tokenStore.AccessToken != tokenBefore) {
                    refreshed = true;
                } else {
                    refreshed = await TryRefreshTokenAsync(cancellationToken);
                }
            } finally {
                RefreshLock.Release();
            }

            if (!refreshed) {
                _tokenStore.ClearTokens();
                _session.RaiseSessionExpired();
                return response;
            }

            request.Headers.Authorization = new AuthenticationHeaderValue("Bearer", _tokenStore.AccessToken);
            return await base.SendAsync(request, cancellationToken);
        }

        return response;
    }

    private async Task<bool> TryRefreshTokenAsync(CancellationToken cancellationToken) {
        if (string.IsNullOrEmpty(_tokenStore.RefreshToken))
            return false;

        try {
            var authResponse = await _authApi.RefreshAsync(_tokenStore.RefreshToken);
            if (authResponse == null) return false;

            _tokenStore.SetTokens(authResponse.Token, authResponse.RefreshToken);
            _session.Set(authResponse.Token, _session.CurrentUser);
            return true;
        } catch {
            return false;
        }
    }
}

using System;
using System.Collections.Generic;
using System.Net;
using System.Net.Http;
using System.Net.Http.Json;
using System.Text.Json;
using System.Threading.Tasks;
using MallClient.Interfaces;
using MallClient.Models;

namespace MallClient.Services;

public abstract class BaseApiService<T> : IApiCrudService<T> where T : class {
    protected readonly IApiClientFactory Clients;
    protected readonly string Endpoint;
    protected readonly ISession Session;

    protected BaseApiService(IApiClientFactory clients, ISession session, string endpoint) {
        Clients = clients;
        Session = session;
        Endpoint = endpoint;
    }

    protected async Task<TR> HandleResponse<TR>(HttpResponseMessage response) {
        if (response.StatusCode == HttpStatusCode.Unauthorized) {
            Session.RaiseSessionExpired();
            throw new ApiException("Требуется авторизация. Войдите снова.", 401);
        }

        if (response.StatusCode == HttpStatusCode.Forbidden) {
            throw new ApiException("Недостаточно прав для этой операции.", 403);
        }

        if (response.StatusCode == HttpStatusCode.NotFound) {
            throw new ApiException("Запись не найдена.", 404);
        }

        if (!response.IsSuccessStatusCode) {
            var error = await response.Content.ReadAsStringAsync();
            try {
                var apiErr = JsonSerializer.Deserialize<ApiErrorResponse>(error);
                if (apiErr != null && !string.IsNullOrEmpty(apiErr.Error))
                    throw new ApiException(apiErr.Error, (int)response.StatusCode);
            } catch (JsonException) { }
            throw new ApiException($"Ошибка сервера: {(int)response.StatusCode} {response.StatusCode}", (int)response.StatusCode);
        }

        if (typeof(TR) == typeof(EmptyResponse))
            return (TR)(object)new EmptyResponse();

        if (response.StatusCode == HttpStatusCode.NoContent) {
            if (typeof(TR) == typeof(List<T>))
                return (TR)(object)new List<T>();
            return default!;
        }

        var content = await response.Content.ReadAsStringAsync();
        if (string.IsNullOrWhiteSpace(content) || content.Trim() == "null") {
            if (typeof(TR) == typeof(List<T>))
                return (TR)(object)new List<T>();
            return default!;
        }

        try {
            var result = JsonSerializer.Deserialize<TR>(content, new JsonSerializerOptions { PropertyNameCaseInsensitive = true });
            if (result == null && typeof(TR) == typeof(List<T>)) {
                return (TR)(object)new List<T>();
            }
            return result!;
        } catch {
            throw new ApiException("Некорректный ответ сервера", (int)response.StatusCode);
        }
    }

    public virtual Task<List<T>> GetAllAsync() => GetAllAsync(100, 0);

    public virtual async Task<List<T>> GetAllAsync(int limit = 100, int offset = 0) {
        var client = Clients.Create();
        var res = await client.GetAsync($"{Endpoint}?limit={limit}&offset={offset}");
        return await HandleResponse<List<T>>(res);
    }

    public virtual async Task CreateAsync(object data) {
        var client = Clients.Create();
        var res = await client.PostAsJsonAsync(Endpoint, data);
        await HandleResponse<EmptyResponse>(res);
    }

    public virtual async Task UpdateAsync(long id, object data) {
        var client = Clients.Create();
        var res = await client.PutAsJsonAsync($"{Endpoint}/{id}", data);
        await HandleResponse<EmptyResponse>(res);
    }

    public virtual async Task DeleteAsync(long id) {
        var client = Clients.Create();
        var res = await client.DeleteAsync($"{Endpoint}/{id}");
        await HandleResponse<EmptyResponse>(res);
    }
}

public class EmptyResponse { }

using System.Text.RegularExpressions;
using MallClient.Interfaces;
using MallClient.Models;

namespace MallClient.Services;

public class InputValidator : IInputValidator {
    private static readonly Regex EmailRegex = new(@"^[^@\s]+@[^@\s]+\.[^@\s]+$");
    private static readonly Regex PhoneRegex = new(@"^\+?[0-9]{7,15}$");
    private static readonly Regex InnRegex = new(@"^\d{10}$|^\d{12}$");
    private static readonly Regex RoomNumRegex = new(@"^[A-Za-zА-Яа-я0-9\-/]{1,20}$");

    public string? ValidateField(FieldItem field, bool isCreate) {
        if (field.MaxLength > 0 && field.Value.Length > field.MaxLength)
            return $"Поле '{field.Label}': максимум {field.MaxLength} символов.";

        if (isCreate && string.IsNullOrWhiteSpace(field.Value) && !field.IsDropdown) {
            return $"Поле '{field.Label}': обязательное поле.";
        }

        if (field.Name == "Email" && !string.IsNullOrWhiteSpace(field.Value) && !EmailRegex.IsMatch(field.Value))
            return $"Поле '{field.Label}': некорректный email.";

        if (field.Name == "ContactPhone" && !string.IsNullOrWhiteSpace(field.Value) && !PhoneRegex.IsMatch(field.Value))
            return $"Поле '{field.Label}': некорректный телефон (от 7 до 15 цифр).";

        if (field.Name == "Inn" && !string.IsNullOrWhiteSpace(field.Value) && !InnRegex.IsMatch(field.Value))
            return $"Поле '{field.Label}': ИНН должен содержать 10 или 12 цифр.";

        if (field.Name == "RoomNumber" && !string.IsNullOrWhiteSpace(field.Value) && !RoomNumRegex.IsMatch(field.Value))
            return $"Поле '{field.Label}': допустимы буквы, цифры, дефис и слэш (1–20 символов).";

        if (!string.IsNullOrWhiteSpace(field.Value)) {
            int len = field.Value.Length;

            if ((field.Name == "Name" || field.Name == "FullName") && len < 2)
                return $"Поле '{field.Label}': минимум 2 символа.";

            if (field.Name == "LegalAddress" && len < 5)
                return $"Поле '{field.Label}': минимум 5 символов.";

            if (field.Name == "Description" && len < 5)
                return $"Поле '{field.Label}': минимум 5 символов.";

            if (field.Name == "RequestType" && len < 2)
                return $"Поле '{field.Label}': минимум 2 символа.";

            if (field.Name == "Type" && len < 1)
                return $"Поле '{field.Label}': минимум 1 символ.";
        }

        if (field.IsNumeric && !string.IsNullOrWhiteSpace(field.Value)) {
            if (double.TryParse(field.Value.Replace(',', '.'), out var num)) {
                if (field.Name == "Floor") {
                    if (num < 0) return "Этаж не может быть отрицательным.";
                    if (num > 100) return "Этаж не может быть больше 100.";
                }

                if (field.Name == "Area") {
                    if (num <= 0) return "Площадь должна быть больше нуля.";
                    if (num > 9999999) return "Площадь не может превышать 9 999 999.";
                }

                if (field.Name == "Amount") {
                    if (num <= 0) return "Сумма должна быть больше нуля.";
                    if (num > 999999999) return "Сумма не может превышать 999 999 999.";
                }

                if (field.Name == "RatePerSqm") {
                    if (num <= 0) return "Ставка должна быть больше нуля.";
                    if (num > 999999) return "Ставка не может превышать 999 999.";
                }

                if (field.Name == "DepositAmount") {
                    if (num < 0) return "Сумма залога не может быть отрицательной.";
                    if (num > 999999999) return "Залог не может превышать 999 999 999.";
                }
            } else {
                return $"Поле '{field.Label}': введите корректное число.";
            }
        }

        return null;
    }
}

using System;
using System.Collections.Generic;
using System.Globalization;
using System.Linq;
using System.Reflection;
using MallClient.Models;

namespace MallClient.Services;

public class PayloadBuilder {
    public Dictionary<string, object> Build<T>(T? entity, IEnumerable<FieldItem> fields) where T : class {
        var payload = new Dictionary<string, object>();

        if (entity != null) {
            foreach (var prop in typeof(T).GetProperties()) {
                if (prop.Name == "Id" || prop.Name == "CreatedAt" || prop.Name == "PasswordHash") continue;

                var val = prop.GetValue(entity);
                if (val != null) {
                    var propName = ToSnakeCase(prop.Name);
                    payload[propName] = val;
                }
            }
        }

        foreach (var field in fields) {
            var prop = typeof(T).GetProperty(field.Name);
            if (prop == null) continue;

            var propName = ToSnakeCase(field.Name);
            var t = prop.PropertyType;
            var raw = field.Value?.Trim() ?? string.Empty;

            if (t == typeof(int) || t == typeof(int?)) {
                if (int.TryParse(raw, NumberStyles.Integer, CultureInfo.InvariantCulture, out var i))
                    payload[propName] = i;
            } else if (t == typeof(long) || t == typeof(long?)) {
                if (long.TryParse(raw, NumberStyles.Integer, CultureInfo.InvariantCulture, out var l))
                    payload[propName] = l;
            } else if (t == typeof(decimal) || t == typeof(decimal?)) {
                if (decimal.TryParse(raw.Replace(',', '.'), NumberStyles.Number, CultureInfo.InvariantCulture, out var m))
                    payload[propName] = m;
            } else if (t == typeof(double) || t == typeof(double?)) {
                if (double.TryParse(raw.Replace(',', '.'), NumberStyles.Float, CultureInfo.InvariantCulture, out var d))
                    payload[propName] = d;
            } else {
                payload[propName] = field.Value ?? string.Empty;
            }
        }

        return payload;
    }

    private string ToSnakeCase(string input) {
        if (string.IsNullOrEmpty(input)) return input;
        return string.Concat(input.Select((x, i) =>
            i > 0 && char.IsUpper(x) ? "_" + x.ToString() : x.ToString())).ToLower();
    }
}

using System;
using MallClient.Interfaces;
using MallClient.Models;

namespace MallClient.Services;

public class Session : ISession {
    public string Token { get; private set; } = string.Empty;
    public User CurrentUser { get; private set; } = new();

    public bool IsAuthenticated => !string.IsNullOrEmpty(Token);
    public bool IsAdmin => CurrentUser.Role == "admin";
    public bool IsManager => CurrentUser.Role == "manager";
    public bool IsAccountant => CurrentUser.Role == "accountant";
    public bool IsTechnician => CurrentUser.Role == "technician";
    public bool IsSupport => CurrentUser.Role == "support";

    public event Action? SessionExpired;

    public void Set(string token, User user) {
        Token = token;
        CurrentUser = user;
    }

    public void Clear() {
        Token = string.Empty;
        CurrentUser = new User();
    }

    public void UpdateFullName(string fullName) {
        CurrentUser.FullName = fullName;
    }

    public void RaiseSessionExpired() {
        SessionExpired?.Invoke();
    }
}

using System;
using System.Windows;
using Microsoft.Extensions.DependencyInjection;
using MallClient.Interfaces;
using MallClient.Models;
using MallClient.ViewModels;
using MallClient.Views;
using MallClient.Schemas;

namespace MallClient.Services;

public class WindowService : IWindowService {
    private readonly IServiceProvider _serviceProvider;
    private readonly SchemaFactory _schemaFactory;

    public WindowService(IServiceProvider serviceProvider, SchemaFactory schemaFactory) {
        _serviceProvider = serviceProvider;
        _schemaFactory = schemaFactory;
    }

    public void ShowLoginWindow() {
        var win = _serviceProvider.GetRequiredService<LoginWindow>();
        Application.Current.MainWindow = win;
        win.Show();
        CloseOtherWindows(win);
    }

    public void ShowRegisterWindow() {
        var win = _serviceProvider.GetRequiredService<RegisterWindow>();
        Application.Current.MainWindow = win;
        win.Show();
        CloseOtherWindows(win);
    }

    public void ShowMainWindow() {
        var win = _serviceProvider.GetRequiredService<MainWindow>();
        Application.Current.MainWindow = win;
        win.Show();
        CloseOtherWindows(win);
    }

    private void CloseOtherWindows(Window currentWindow) {
        foreach (Window window in Application.Current.Windows) {
            if (window == currentWindow) continue;

            if (window is LoginWindow lw) lw.IsSwitchingWindow = true;
            if (window is RegisterWindow rw) rw.IsSwitchingWindow = true;

            if (window.IsLoaded) window.Close();
        }
    }

    public bool ShowEditWindow<T>(T? entity, string title) where T : class {
        var service = GetCrudService<T>();
        var validator = _serviceProvider.GetRequiredService<IInputValidator>();
        var payloadBuilder = _serviceProvider.GetRequiredService<PayloadBuilder>();
        var schema = _schemaFactory.GetSchema<T>();

        var vm = new GenericEditViewModel<T>(
            entity,
            service,
            validator,
            payloadBuilder,
            schema,
            _serviceProvider
        );

        var win = _serviceProvider.GetRequiredService<GenericEditWindow>();
        win.DataContext = vm;
        win.Title = title;

        vm.CloseAction = () => win.DialogResult = true;

        return win.ShowDialog() == true;
    }

    public bool ShowConfirmWindow(string message) {
        var win = new ConfirmWindow(message) {
            Owner = Application.Current.MainWindow
        };
        return win.ShowDialog() == true;
    }

    private IApiCrudService<T> GetCrudService<T>() where T : class {
        return typeof(T).Name switch {
            nameof(Premise) => (IApiCrudService<T>)(object)_serviceProvider.GetRequiredService<IPremiseService>(),
            nameof(Tenant) => (IApiCrudService<T>)(object)_serviceProvider.GetRequiredService<ITenantService>(),
            nameof(LeaseContract) => (IApiCrudService<T>)(object)_serviceProvider.GetRequiredService<IContractService>(),
            nameof(Payment) => (IApiCrudService<T>)(object)_serviceProvider.GetRequiredService<IPaymentService>(),
            nameof(ServiceRequest) => (IApiCrudService<T>)(object)_serviceProvider.GetRequiredService<IRequestService>(),
            nameof(User) => (IApiCrudService<T>)(object)_serviceProvider.GetRequiredService<IUserService>(),
            _ => throw new NotSupportedException($"Для типа {typeof(T).Name} не настроен IApiCrudService")
        };
    }
}

using System;
using System.Collections.Generic;
using System.Collections.ObjectModel;
using System.Linq;
using System.Reflection;
using System.Threading.Tasks;
using System.Windows.Input;
using Microsoft.Extensions.DependencyInjection;
using MallClient.Interfaces;
using MallClient.Models;
using MallClient.Schemas;
using MallClient.Services;

namespace MallClient.ViewModels;

public class GenericEditViewModel<T> : BaseViewModel, IEditViewModel where T : class {
    private readonly IApiCrudService<T> _service;
    private readonly IInputValidator _validator;
    private readonly PayloadBuilder _payloadBuilder;
    private readonly IEntitySchema _schema;
    private readonly IServiceProvider _serviceProvider;
    private readonly T? _entity;
    private readonly long _entityId;

    private string _errorMessage = string.Empty;
    public string ErrorMessage { get => _errorMessage; set { _errorMessage = value; OnPropertyChanged(); } }

    public ObservableCollection<FieldItem> Fields { get; } = new();

    private bool _canSave = true;
    public bool CanSave { get => _canSave; set { _canSave = value; OnPropertyChanged(); } }

    public ICommand SaveCommand { get; }
    public bool IsSaved { get; private set; }
    public Action? CloseAction { get; set; }

    public GenericEditViewModel(
        T? entity,
        IApiCrudService<T> service,
        IInputValidator validator,
        PayloadBuilder payloadBuilder,
        IEntitySchema schema,
        IServiceProvider serviceProvider) {
        _entity = entity;
        _service = service;
        _validator = validator;
        _payloadBuilder = payloadBuilder;
        _schema = schema;
        _serviceProvider = serviceProvider;

        SaveCommand = new RelayCommand(async _ => await SaveAsync(), _ => CanSave);

        if (_entity != null) {
            var idProp = typeof(T).GetProperty("Id");
            _entityId = (long)(idProp?.GetValue(_entity) ?? 0);
        }

        LoadFields();
    }

    private void LoadFields() {
        foreach (var prop in typeof(T).GetProperties()) {
            if (prop.Name == "Id" || prop.Name == "PasswordHash" || prop.Name == "CreatedAt") continue;

            var val = _entity != null ? prop.GetValue(_entity)?.ToString() : string.Empty;
            var field = new FieldItem { Name = prop.Name, Label = _schema.GetLabel(prop.Name), Value = val ?? string.Empty };

            var t = prop.PropertyType;
            var fkAttr = prop.GetCustomAttribute<ForeignKeyAttribute>();

            if (fkAttr != null) {
                field.IsDropdown = true;
                field.IsNumeric = false;
            } else if (t == typeof(int) || t == typeof(long) || t == typeof(double) || t == typeof(decimal)) {
                field.IsNumeric = true;
                if (field.MaxLength == 0) field.MaxLength = 10;
            }

            if (prop.Name.EndsWith("Date", StringComparison.OrdinalIgnoreCase)) {
                field.IsDate = true;
                field.IsNumeric = false;
            }

            _schema.Configure(field);
            _schema.ConfigureDropdowns(field);

            Fields.Add(field);

            if (fkAttr != null) {
                _ = LoadDropdownOptionsAsync(field, fkAttr.ProviderType);
            }
        }
    }

    private async Task LoadDropdownOptionsAsync(FieldItem field, Type providerType) {
        field.Options.Add(new OptionItem { Value = string.Empty, DisplayName = "Загрузка..." });

        try {
            var provider = (IOptionsProvider)_serviceProvider.GetRequiredService(providerType);
            var options = await provider.GetOptionsAsync();

            field.Options.Clear();
            foreach (var opt in options) {
                field.Options.Add(opt);
            }

            if (string.IsNullOrEmpty(field.Value) && field.Options.Any()) {
                field.Value = field.Options.First().Value;
            }
        } catch (Exception ex) {
            field.Options.Clear();
            field.Options.Add(new OptionItem { Value = "", DisplayName = $"Ошибка загрузки: {ex.Message}" });
            System.Diagnostics.Debug.WriteLine($"[FK ERROR] {ex}");
        }
    }

    private async Task SaveAsync() {
        try {
            ErrorMessage = string.Empty;
            bool isCreate = _entity == null;

            foreach (var field in Fields) {
                var error = _validator.ValidateField(field, isCreate);
                if (error != null) {
                    ErrorMessage = error;
                    return;
                }
            }

            var payload = _payloadBuilder.Build(_entity, Fields);
            if (isCreate) await _service.CreateAsync(payload);
            else await _service.UpdateAsync(_entityId, payload);

            IsSaved = true;
            CloseAction?.Invoke();
        } catch (Exception ex) {
            ErrorMessage = ex.Message;
        }
    }
}

using System;
using System.Threading.Tasks;
using System.Windows.Input;
using MallClient.Interfaces;

namespace MallClient.ViewModels;

public class LoginViewModel : BaseViewModel {
    private readonly IAuthService _auth;
    private readonly IWindowService _windows;

    private string _email = string.Empty;
    private string _password = string.Empty;
    private string _error = string.Empty;

    public string Email { get => _email; set { _email = value; OnPropertyChanged(); } }
    public string Password { get => _password; set { _password = value; OnPropertyChanged(); } }
    public string Error { get => _error; set { _error = value; OnPropertyChanged(); } }

    public ICommand LoginCommand { get; }
    public ICommand GoToRegisterCommand { get; }

    public LoginViewModel(IAuthService auth, IWindowService windows) {
        _auth = auth;
        _windows = windows;
        LoginCommand = new RelayCommand(async _ => await LoginAsync());
        GoToRegisterCommand = new RelayCommand(_ => _windows.ShowRegisterWindow());
    }

    private async Task LoginAsync() {
        try {
            if (string.IsNullOrWhiteSpace(Email) || string.IsNullOrWhiteSpace(Password)) {
                Error = "Заполните все поля";
                return;
            }

            var success = await _auth.LoginAsync(Email, Password);
            if (!success) {
                Error = "Неверный email или пароль";
                return;
            }

            _windows.ShowMainWindow();
        } catch (Exception ex) {
            Error = $"Ошибка входа: {ex.Message}";
        }
    }
}

    public bool CanCreateRequest => _session.IsAdmin || _session.IsSupport;

    public MainViewModel(
        ISession session,

        IWindowService windows,
        IPremiseService premises,
        ITenantService tenants,

        IPaymentService payments,
        IRequestService requests,

        _windows = windows;
        _premises = premises;
        _tenants = tenants;
        _contracts = contracts;
        _payments = payments;
        _requests = requests;
        _users = users;

        LogoutCommand = new RelayCommand(_ => Logout());
    }

    private void Logout() {
        try { _auth.Logout(); } catch { }

        Premises.Clear();
        Tenants.Clear();
        Contracts.Clear();
        Payments.Clear();
        Requests.Clear();
        Users.Clear();
        ErrorMessage = string.Empty;

        _windows.ShowLoginWindow();
    }

    public async Task LoadDataAsync() {
        var errors = new List<string>();

        if (CanViewPremises) await SafeLoad(
            async () => { var data = await _premises.GetAllAsync(); Premises.Clear(); foreach (var i in data) Premises.Add(i); },
            "помещения", errors);

        if (CanViewTenants) await SafeLoad(
            async () => { var data = await _tenants.GetAllAsync(); Tenants.Clear(); foreach (var i in data) Tenants.Add(i); },
            "арендаторы", errors);

        if (CanViewContracts) await SafeLoad(
            async () => { var data = await _contracts.GetAllAsync(); Contracts.Clear(); foreach (var i in data) Contracts.Add(i); },
            "договоры", errors);

        if (CanViewPayments) await SafeLoad(
            async () => { var data = await _payments.GetAllAsync(); Payments.Clear(); foreach (var i in data) Payments.Add(i); },
            "платежи", errors);

        if (CanViewRequests) await SafeLoad(
            async () => { var data = await _requests.GetAllAsync(); Requests.Clear(); foreach (var i in data) Requests.Add(i); },
            "заявки", errors);

        if (CanViewUsers) await SafeLoad(
            async () => { var data = await _users.GetAllAsync(); Users.Clear(); foreach (var i in data) Users.Add(i); },
            "пользователи", errors);

        ErrorMessage = errors.Count == 0 ? string.Empty : string.Join("\n", errors);
    }

    private static async Task SafeLoad(Func<Task> action, string title, List<string> errors) {
        try {
            await action();
        } catch (ApiException ex) when (ex.StatusCode == 401) {
        } catch (ApiException ex) when (ex.StatusCode == 403) {
            errors.Add($"{title}: недостаточно прав");
        } catch (Exception ex) {
            errors.Add($"{title}: {ex.Message}");
        }
    }

    private async Task ReloadAsync<T>(Func<Task<List<T>>> loader, ObservableCollection<T> target, string title) {
        try {
            var data = await loader();
            target.Clear();
            foreach (var i in data) target.Add(i);
            ErrorMessage = string.Empty;
        } catch (ApiException ex) when (ex.StatusCode == 401) {
        } catch (Exception ex) {
            ErrorMessage = $"{title}: {ex.Message}";
        }
    }

    public Task ReloadPremisesAsync() => CanViewPremises

    public Task ReloadTenantsAsync() => CanViewTenants

using System;
using System.Threading.Tasks;

using MallClient.Interfaces;

namespace MallClient.ViewModels;

public class RegisterViewModel : BaseViewModel {
    private readonly IAuthService _auth;
    private readonly IWindowService _windows;

    private string _email = string.Empty;
    private string _password = string.Empty;
    private string _fullName = string.Empty;
    private string _error = string.Empty;

    public string Email { get => _email; set { _email = value; OnPropertyChanged(); } }
    public string Password { get => _password; set { _password = value; OnPropertyChanged(); } }
    public string FullName { get => _fullName; set { _fullName = value; OnPropertyChanged(); } }

using System;
using System.Windows.Input;

namespace MallClient.ViewModels;

public class RelayCommand : ICommand {
    private readonly Action<object?> _execute;
    private readonly Func<object?, bool>? _canExecute;

    public RelayCommand(Action<object?> execute, Func<object?, bool>? canExecute = null) {
        _execute = execute;
        _canExecute = canExecute;
    }

    public bool CanExecute(object? parameter) => _canExecute == null || _canExecute(parameter);
    public void Execute(object? parameter) => _execute(parameter);

    public event EventHandler? CanExecuteChanged {
        add => CommandManager.RequerySuggested += value;
        remove => CommandManager.RequerySuggested -= value;
    }

    public static void RaiseCanExecuteChanged() => CommandManager.InvalidateRequerySuggested();
}

using System.Text.RegularExpressions;
using System.Windows;
using System.Windows.Controls;
using System.Windows.Input;
using MallClient.Models;

namespace MallClient.Views;

public partial class GenericEditWindow : Window {
    private static readonly Regex NumericRegex = new(@"^[0-9]*[.,]?[0-9]*$");
    private static readonly Regex PhoneRegex = new(@"^\+?[0-9]*$");

    public GenericEditWindow() {
        InitializeComponent();
    }

    private void TitleBar_MouseLeftButtonDown(object sender, MouseButtonEventArgs e) {
        if (e.ButtonState == MouseButtonState.Pressed)
            DragMove();
    }

    private void CloseButton_Click(object sender, RoutedEventArgs e) {
        DialogResult = false;
    }

    private void NumericTextBox_PreviewTextInput(object sender, TextCompositionEventArgs e) {
        if (sender is not TextBox tb) return;
        if (tb.DataContext is not FieldItem field) return;

        if (field.MaxLength > 0 && tb.Text.Length >= field.MaxLength) {
            e.Handled = true;
            return;
        }

        var proposed = tb.Text.Insert(tb.CaretIndex, e.Text);
        if (!NumericRegex.IsMatch(proposed))
            e.Handled = true;
    }

    private void NumericTextBox_Pasting(object sender, DataObjectPastingEventArgs e) {
        if (e.DataObject.GetDataPresent(typeof(string))) {
            var text = (string)e.DataObject.GetData(typeof(string));
            if (!NumericRegex.IsMatch(text))
                e.CancelCommand();
        } else {
            e.CancelCommand();
        }
    }

    private void PhoneTextBox_PreviewTextInput(object sender, TextCompositionEventArgs e) {
        if (sender is not TextBox tb) return;
        if (tb.DataContext is not FieldItem field) return;

        if (field.MaxLength > 0 && tb.Text.Length >= field.MaxLength) {
            e.Handled = true;
            return;
        }

        var proposed = tb.Text.Insert(tb.CaretIndex, e.Text);
        if (!PhoneRegex.IsMatch(proposed))
            e.Handled = true;
    }
}
